### Store

// stores/authStore.ts
import { defineStore } from 'pinia'
import { ref, computed } from 'vue'

export const useAuthStore = defineStore('auth', () => {
  const accessToken = ref(localStorage.getItem('accessToken') || '')
  const refreshToken = ref(localStorage.getItem('refreshToken') || '')
  const user = ref<UserProfile | null>(null)

  const isLoggedIn = computed(() => !!accessToken.value)
  const role = computed(() => user.value?.role ?? 0)
  const isAnchor = computed(() => role.value >= 2)  // 2=ANCHOR 3=ADMIN

  function setTokens(loginRes: LoginResponse) {
    accessToken.value = loginRes.accessToken
    refreshToken.value = loginRes.refreshToken
    localStorage.setItem('accessToken', loginRes.accessToken)
    localStorage.setItem('refreshToken', loginRes.refreshToken)
  }

  function setUser(u: UserProfile) { user.value = u }
  function clearTokens() {
    accessToken.value = ''
    refreshToken.value = ''
    user.value = null
    localStorage.removeItem('accessToken')
    localStorage.removeItem('refreshToken')
  }

  return { accessToken, refreshToken, user, isLoggedIn, role, isAnchor,
           setTokens, setUser, clearTokens }
})
```

// composables/useAuth.ts
export function useAuth() {
  const authStore = useAuthStore()
  const http = useHttp()  // axios 实例

  async function login(req: LoginRequest) {
    const res = await http.post<LoginResponse>('/auth/login', req, {
      headers: { 'X-Device-Id': getDeviceId() }
    })
    authStore.setTokens(res.data)
    return res.data
  }

  async function register(req: RegisterRequest) {
    return http.post('/auth/register', req)
  }

  async function logout() {
    await http.post('/auth/logout', { refreshToken: authStore.refreshToken })
    authStore.clearTokens()
  }

  async function refresh() {
    const res = await http.post<LoginResponse>('/auth/refresh', {
      refreshToken: authStore.refreshToken
    })
    authStore.setTokens(res.data)
  }

  return { login, register, logout, refresh }
}
```

### Store

```typescript
// stores/seckillStore.ts
export const useSeckillStore = defineStore('seckill', () => {
  const activities = ref<SeckillActivity[]>([])
  const currentActivity = ref<SeckillActivity | null>(null)
  const orders = ref<SeckillOrder[]>([])

  // 秒杀状态机
  const seckillStatus = computed<SeckillStatus>(() => {
    if (!currentActivity.value) return 'PENDING'
    const a = currentActivity.value
    const start = new Date(a.startTime).getTime()
    const end = new Date(a.endTime).getTime()
    const now = serverNow()
    if (now < start) return 'PENDING'
    if (now < start + 300000) return 'READY'
    if (now >= start && now <= end && a.status === 1) return 'PROCESSING'
    if (a.status === 2 || a.status === 3) return 'FAILED'
    return 'PENDING'
  })

  // 秒杀倒计时（ms）
  const countdown = computed(() => {
    if (!currentActivity.value) return 0
    const start = new Date(currentActivity.value.startTime).getTime()
    return Math.max(0, start - serverNow())
  })

  // orderNo 匹配器
  const orderCallbacks = new Map<string, (msg: SecKillResultData) => void>()

  function registerOrderCallback(orderNo: string, cb: (msg: SecKillResultData) => void, timeout = 15000) {
    const timer = setTimeout(() => {
      orderCallbacks.delete(orderNo)
      cb({ orderNo, ok: false, reason: 'timeout' })
    }, timeout)
    orderCallbacks.set(orderNo, (msg) => { clearTimeout(timer); cb(msg) })
  }

  function resolveOrderCallback(msg: { orderNo: string; ok: boolean; reason: string }) {
    const cb = orderCallbacks.get(msg.orderNo)
    if (cb) { cb(msg); orderCallbacks.delete(msg.orderNo) }
  }

  async function fetchActivities(roomId?: number) {
    const params = roomId ? { roomId } : {}
    const res = await api.get<SeckillActivity[]>('/seckill/activity/list', { params })
    activities.value = res.data
  }

  async function placeOrder(activityId: number) {
    return api.post<PlaceOrderResponse>('/seckill/order', { activityId })
  }

  async function fetchOrders() {
    const res = await api.get<SeckillOrder[]>('/seckill/order/list')
    orders.value = res.data
  }

  async function cancelOrder(orderNo: string) {
    await api.put(`/seckill/order/${orderNo}/cancel`)
    await fetchOrders()
  }

  return { activities, currentActivity, orders, seckillStatus, countdown,
           orderCallbacks, registerOrderCallback, resolveOrderCallback,
           fetchActivities, placeOrder, fetchOrders, cancelOrder }
})
```

### Composable: useWebSocket

```typescript
// composables/useWebSocket.ts
export function useWebSocket(roomId: number) {
  const ws = ref<WebSocket | null>(null)
  const connected = ref(false)
  const displayName = ref('')
  const online = ref(0)
  const msgList = ref<WsBarrage['data'][]>([])
  let pingTimer: number | undefined
  let reconnectTimer: number | undefined
  let retryCount = 0

  function connect(token?: string) {
    const params = new URLSearchParams({ deviceId: getDeviceId() })
    if (token) params.set('token', token)

    ws.value = new WebSocket(`${import.meta.env.VITE_WS_HOST}/ws/live/${roomId}?${params}`)

    ws.value.onopen = () => {
      connected.value = true
      retryCount = 0
      startPing()
    }

    ws.value.onmessage = (e: MessageEvent) => {
      const msg: WsServerMessage = JSON.parse(e.data)
      routeMessage(msg)
    }

    ws.value.onclose = () => {
      connected.value = false
      stopPing()
      scheduleReconnect()
    }
  }

  function routeMessage(msg: WsServerMessage) {
    switch (msg.type) {
      case 'CONNECTED':
        displayName.value = msg.data.displayName
        online.value = msg.data.online
        break
      case 'PONG':
        break
      case 'BARRAGE':
        msgList.value.push(msg.data)
        break
      case 'GIFT':
        msgList.value.push(msg.data)           // 聊天面板
        triggerGiftEffect(msg.data)            // 全屏特效
        refreshLeaderboard()                   // 刷新排行榜
        break
      case 'SEC_KILL_RESULT':
        seckillStore.resolveOrderCallback(msg.data)
        break
      case 'ROOM_CLOSED':
        showToast('直播已结束')
        break
      case 'KICK':
        handleKick()
        break
      case 'NEED_AUTH':
        showToast('请先登录后再操作')
        break
    }
  }

  function sendAuth(token: string) {
    send({ type: 'AUTH', data: { token } })
  }

  function sendBarrage(content: string) {
    send({ type: 'BARRAGE', data: { content } })
  }

  function sendGift(giftId: number, quantity = 1) {
    send({ type: 'GIFT', data: { giftId, quantity } })
  }

  // 消息分发: 不在此文件中混合所有逻辑，抛出事件让组件订阅
  const onMessage = createEventHook<WsServerMessage>()
  const onKick = createEventHook<void>()

  function startPing() {
    pingTimer = window.setInterval(() => send({ type: 'PING' }), 30000)
  }
  function stopPing() { clearInterval(pingTimer) }
  function scheduleReconnect() {
    if (retryCount >= 5) return
    const delay = Math.min(1000 * 2 ** retryCount, 10000)
    reconnectTimer = window.setTimeout(() => { retryCount++; connect() }, delay)
  }

  return { ws, connected, displayName, online, msgList,
           connect, sendAuth, sendBarrage, sendGift,
           onMessage, onKick }
}
```

### Store

```typescript
// stores/pusherStore.ts
export const usePusherStore = defineStore('pusher', () => {
  const status = ref<PushStatus>('IDLE')
  const pushDuration = ref(0)        // 开播时长(s)
  const localStream = ref<MediaStream | null>(null)
  const devices = ref<CameraDevice[]>([])
  const currentDeviceId = ref('')
  const mirrored = ref(true)          // 本地预览是否镜像
  const sessionId = ref('')

  let pushTimer: number | undefined

  async function initCamera(facingMode: 'user' | 'environment' = 'user') {
    status.value = 'CAMERA_INIT'
    const stream = await navigator.mediaDevices.getUserMedia({
      video: { width: 1280, height: 720, facingMode },
      audio: { echoCancellation: true, noiseSuppression: true }
    })
    localStream.value = stream

    const deviceList = await navigator.mediaDevices.enumerateDevices()
    devices.value = deviceList
      .filter(d => d.kind === 'videoinput')
      .map(d => ({
        deviceId: d.deviceId,
        label: d.label || `Camera ${devices.value.length + 1}`,
        facing: facingMode
      }))
    currentDeviceId.value = devices.value[0]?.deviceId || ''
    status.value = 'READY'
  }

  async function switchCamera() {
    if (!localStream.value) return
    const newFacing = currentDeviceId.value.includes('environment') ? 'user' : 'environment'
    stopTracks()
    await initCamera(newFacing)
  }

  function stopTracks() {
    localStream.value?.getTracks().forEach(tr => tr.stop())
    localStream.value = null
  }

  function startTimer() {
    pushDuration.value = 0
    pushTimer = window.setInterval(() => pushDuration.value++, 1000)
  }

  function stopTimer() {
    clearInterval(pushTimer)
    pushDuration.value = 0
  }

  function reset() {
    stopTracks()
    stopTimer()
    status.value = 'IDLE'
    sessionId.value = ''
  }

  return { status, pushDuration, localStream, devices, currentDeviceId,
           mirrored, sessionId, initCamera, switchCamera, stopTracks,
           startTimer, stopTimer, reset }
})
```

### Composable: useWebRTC

```typescript
// composables/useWebRTC.ts
export function useWebRTC() {
  const pusherStore = usePusherStore()
  let pc: RTCPeerConnection | null = null

  const iceServers: RTCConfiguration = {
    iceServers: [
      { urls: 'stun:stun.l.google.com:19302' },
      // TURN 服务器（生产必配）
      // { urls: 'turn:turn.example.com:3478', username: '...', credential: '...' }
    ]
  }

  async function startPush(roomId: number) {
    if (!pusherStore.localStream) throw new Error('摄像头未初始化')
    pusherStore.status = 'CONNECTING'

    pc = new RTCPeerConnection(iceServers)

    // 添加音视频轨道
    pusherStore.localStream.getTracks().forEach(tr => pc!.addTrack(tr, pusherStore.localStream!))

    // ICE 候选 → 发送到信令
    pc.onicecandidate = async (ev) => {
      if (ev.candidate) {
        await sendTrickle(ev.candidate.toJSON())
      }
    }

    // 连接状态变化
    pc.onconnectionstatechange = () => {
      if (pc?.connectionState === 'failed' || pc?.connectionState === 'disconnected') {
        stopPush()
        pusherStore.status = 'ERROR'
      }
    }

    // 创建 SDP Offer
    const offer = await pc.createOffer({
      offerToReceiveAudio: false,
      offerToReceiveVideo: false
    })
    await pc.setLocalDescription(offer)

    // 通过 WHEP 信令发送 Offer → 接收 Answer
    const whipRes = await http.post<WhipSignalResponse>('/live/webrtc/push', {
      sdp: offer.sdp,
      roomId
    }, { headers: { 'Content-Type': 'application/sdp' } })

    await pc.setRemoteDescription(
      new RTCSessionDescription({ type: 'answer', sdp: whipRes.data.sdp })
    )

    pusherStore.sessionId = whipRes.data.sessionId
    pusherStore.status = 'PUSHING'
    pusherStore.startTimer()

    // 通知后端直播间已绑定流
    await http.post('/live/room/start', { roomId, sessionId: whipRes.data.sessionId })

    return whipRes.data
  }

  async function stopPush() {
    if (pc) { pc.close(); pc = null }

    // 通知信令关闭推流会话
    if (pusherStore.sessionId) {
      await http.delete(`/live/webrtc/push/${pusherStore.sessionId}`).catch(() => {})
    }

    pusherStore.stopTimer()
    pusherStore.status = 'IDLE'
  }

  async function sendTrickle(candidate: RTCIceCandidateInit) {
    await http.post('/live/webrtc/trickle', {
      sessionId: pusherStore.sessionId,
      candidate
    }).catch(() => {})
  }

  onBeforeUnmount(() => {
    stopPush()
    pusherStore.stopTracks()
  })

  return { startPush, stopPush }
}
```
